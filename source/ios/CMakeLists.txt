# Builds a Kayte program into an iOS app (.app) - driven by
# scripts/build-kayte-ios.sh, which also explains the cache variables.
#
# The program is the C that `kayte --native app.kayte -o app.c` writes
# (it #includes source/native/kayte_native_rt.c). iOS apps can't dlopen a
# library from outside their bundle, so instead of loading libkayte_qt6
# at run time the Qt shim (source/qt6/kayte_qt6.cpp) is compiled into the
# app (KAYTE_QT_STATIC), together with Qt's static iOS libraries. Qt's
# CMake takes care of the iOS entry point, Info.plist and QML plugins.
#
# Configure with Qt for iOS's qt-cmake (e.g. ~/Qt/6.11.1/ios/bin/qt-cmake)
# and the Xcode generator.

cmake_minimum_required(VERSION 3.21)
project(KayteIosApp LANGUAGES C CXX)

set(KAYTE_PROGRAM_C "" CACHE FILEPATH "C generated by kayte --native ... -o <name>.c")
set(KAYTE_APP_NAME "KayteApp" CACHE STRING "App (and executable) name")
set(KAYTE_BUNDLE_ID "org.kayte.app" CACHE STRING "Bundle identifier")
set(KAYTE_STAGE_DIR "" CACHE PATH "Files to bundle, laid out as the program's relative paths expect")

if(NOT EXISTS "${KAYTE_PROGRAM_C}")
  message(FATAL_ERROR "KAYTE_PROGRAM_C must point to the generated program C")
endif()
if(NOT IS_DIRECTORY "${KAYTE_STAGE_DIR}")
  message(FATAL_ERROR "KAYTE_STAGE_DIR must be a directory")
endif()

find_package(Qt6 REQUIRED COMPONENTS Widgets)
find_package(Qt6 QUIET COMPONENTS Qml Quick QuickWidgets)
set(KAYTE_WITH_QML OFF)
if(Qt6Qml_FOUND AND Qt6Quick_FOUND AND Qt6QuickWidgets_FOUND)
  set(KAYTE_WITH_QML ON)
endif()
message(STATUS "Kayte iOS app: QML support ${KAYTE_WITH_QML}")

get_filename_component(KAYTE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)

qt_add_executable(kayte_app MACOSX_BUNDLE
  "${KAYTE_PROGRAM_C}"
  "${KAYTE_SOURCE}/qt6/kayte_qt6.cpp")

set_target_properties(kayte_app PROPERTIES
  OUTPUT_NAME "${KAYTE_APP_NAME}"
  C_STANDARD 11
  CXX_STANDARD 17
  MACOSX_BUNDLE_BUNDLE_NAME "${KAYTE_APP_NAME}"
  MACOSX_BUNDLE_GUI_IDENTIFIER "${KAYTE_BUNDLE_ID}"
  XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER "${KAYTE_BUNDLE_ID}"
  # qmlimportscanner also scans the staged files for the QML modules to
  # link in (by default it only scans this directory).
  QT_QML_IMPORT_SCANNER_EXTRA_ROOT_PATHS "${KAYTE_STAGE_DIR}")

target_include_directories(kayte_app PRIVATE "${KAYTE_SOURCE}/native")
target_compile_definitions(kayte_app PRIVATE KAYTE_QT_STATIC)
target_link_libraries(kayte_app PRIVATE Qt6::Widgets)
if(KAYTE_WITH_QML)
  target_compile_definitions(kayte_app PRIVATE KAYTE_QT6_QML)
  target_link_libraries(kayte_app PRIVATE Qt6::Qml Qt6::Quick Qt6::QuickWidgets)
endif()

# QT "loadform" reads Designer .ui files too when UiTools is available.
find_package(Qt6 QUIET COMPONENTS UiTools)
if(Qt6UiTools_FOUND)
  target_compile_definitions(kayte_app PRIVATE KAYTE_QT6_UITOOLS)
  target_link_libraries(kayte_app PRIVATE Qt6::UiTools)
endif()

# Bundle the staged files at the same relative paths: the runtime starts
# the program in the bundle's resource directory, so QML "load",
# "examples/qml/todo.qml" finds what was staged as examples/qml/todo.qml.
file(GLOB_RECURSE KAYTE_RESOURCES RELATIVE "${KAYTE_STAGE_DIR}" "${KAYTE_STAGE_DIR}/*")
foreach(rel IN LISTS KAYTE_RESOURCES)
  get_filename_component(dir "${rel}" DIRECTORY)
  set(file "${KAYTE_STAGE_DIR}/${rel}")
  set_source_files_properties("${file}" PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/${dir}")
  target_sources(kayte_app PRIVATE "${file}")
endforeach()

# Static Qt: link the QML modules the bundled .qml files import.
if(KAYTE_WITH_QML)
  qt_import_qml_plugins(kayte_app)
endif()
